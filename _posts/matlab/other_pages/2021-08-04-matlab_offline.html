% تعداد سایت‌ها و مناطق n = 10; m = 15; % ماتریس فاصله (Distance) d = [ 12 15 20 25 18 22 30 28 35 40 45 50 55 60 65; 10 12 18 22 15 20 25 27 33 38 43 48 52 58 62; 20 18 15 10 12 14 18 20 22 25 28 30 35 38 40; 25 22 18 15 10 12 16 18 20 23 26 28 32 35 38; 30 28 25 20 18 15 12 10 15 18 22 25 28 30 33; 35 30 28 25 22 20 18 15 12 10 15 18 22 25 28; 40 35 30 28 25 22 20 18 15 12 10 14 18 20 22; 45 40 35 30 28 25 22 20 18 15 12 10 15 18 20; 50 45 40 35 30 28 25 22 20 18 15 12 10 14 18; 55 50 45 40 35 30 28 25 22 20 18 15 12 10 14 ]; % وزن جمعیت مناطق (Population Weight) w = [8 10 12 15 9 11 14 13 10 9 7 6 5 4 3]; % هزینه ایجاد مرکز در سایت‌ها (Cost) c = [500 450 480 520 470 490 510 530 550 600]; % پارامتر وزن‌دهی تابع هدف lambda = 0.5; % تنظیمات الگوریتم شبیه‌سازی تبرید maxIter = 1000; T_init = 100; T_min = 1e-3; alpha = 0.95; % راه‌حل اولیه تصادفی currentSolution = randi([0 1],1,n); currentCost = objectiveFunc(currentSolution, d, w, c, lambda); bestSolution = currentSolution; bestCost = currentCost; T = T_init; % آرایه برای ذخیره بهترین مقدار تابع هدف در هر تکرار bestCostHistory = zeros(maxIter,1); for iter = 1:maxIter % تولید همسایه با تغییر یک بیت تصادفی neighbor = currentSolution; idx = randi(n); neighbor(idx) = 1 - neighbor(idx); neighborCost = objectiveFunc(neighbor, d, w, c, lambda); delta = neighborCost - currentCost; % پذیرش همسایه بر اساس قانون تبرید if delta < 0 || rand < exp(-delta/T) currentSolution = neighbor; currentCost = neighborCost; if currentCost < bestCost bestSolution = currentSolution; bestCost = currentCost; end end % ذخیره بهترین هزینه در این تکرار bestCostHistory(iter) = bestCost; % کاهش دما T = T * alpha; if T < T_min bestCostHistory = bestCostHistory(1:iter); % کوتاه کردن آرایه break; end % نمایش پیشرفت هر 100 تکرار if mod(iter,100) == 0 disp(['Iteration ', num2str(iter), ': Best Cost = ', num2str(bestCost)]); end end disp('سایت‌های انتخاب شده (SA):'); disp(find(bestSolution==1)); disp(['بهترین هزینه (SA): ', num2str(bestCost)]); % رسم نمودار پیشرفت بهترین مقدار تابع هدف figure; plot(1:length(bestCostHistory), bestCostHistory, '-b', 'LineWidth', 2); xlabel('تکرار (Iteration)'); ylabel('بهترین مقدار تابع هدف (Best Cost)'); title('پیشرفت الگوریتم شبیه‌سازی تبرید'); grid on; % --- تابع هدف --- function cost = objectiveFunc(solution, d, w, c, lambda) selected = find(solution == 1); if isempty(selected) cost = inf; return; end dist_min = min(d(selected, :), [], 1); cost = sum(w .* dist_min) + lambda * sum(c(selected)); end
